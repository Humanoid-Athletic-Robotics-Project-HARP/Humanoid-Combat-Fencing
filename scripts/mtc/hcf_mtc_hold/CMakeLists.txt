cmake_minimum_required(VERSION 3.16)
project(hcf_mtc_hold)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(pluginlib REQUIRED)
find_package(mujoco_sim_ros2 REQUIRED)
find_package(mujoco_ament REQUIRED)
add_library(hcf_mtc_hold SHARED src/hold_plugin.cpp)
target_compile_features(hcf_mtc_hold PUBLIC cxx_std_17)
ament_target_dependencies(hcf_mtc_hold rclcpp pluginlib mujoco_sim_ros2 mujoco_ament)
pluginlib_export_plugin_description_file(mujoco_sim_ros2 plugins.xml)
install(TARGETS hcf_mtc_hold LIBRARY DESTINATION lib)
ament_package()
